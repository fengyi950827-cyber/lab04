# 載入所需套件
library(haven)
library(ggplot2)
library(ggExtra)
library(ggpubr)

# 讀取 PISA 2022 資料
pisa.dat <- read_sav("PISA_tawian2022_trimmed_lab.sav")

1. ESCS 與 PV1MATH

1.1 繪製散佈圖

# 移除 ESCS 與 PV1MATH 的遺漏值
data1 <- na.omit(pisa.dat[, c("ESCS", "PV1MATH")])

# 建立基本散佈圖
p1 <- ggplot(data1, aes(x = ESCS, y = PV1MATH)) +
  geom_point() +
  labs(
    x = "ESCS",
    y = "PV1MATH"
  )

# 加入 X 軸和 Y 軸的邊際直方圖
ggMarginal(p1, type = "histogram")

# 繪製包含迴歸線和 Pearson 相關係數的散佈圖
ggscatter(
  data1,
  x = "ESCS",
  y = "PV1MATH",
  add = "reg.line",
  conf.int = TRUE,
  cor.coef = TRUE,
  cor.method = "pearson",
  xlab = "ESCS",
  ylab = "PV1MATH"
)

1.2 計算相關係數

# 計算 Pearson 相關係數
cor(data1$ESCS, data1$PV1MATH, method = "pearson")
## [1] 0.3927708
# Pearson 相關係數顯著性檢定
cor.test(
  data1$ESCS,
  data1$PV1MATH,
  method = "pearson"
)
## 
##  Pearson's product-moment correlation
## 
## data:  data1$ESCS and data1$PV1MATH
## t = 31.952, df = 5597, p-value < 2.2e-16
## alternative hypothesis: true correlation is not equal to 0
## 95 percent confidence interval:
##  0.3703870 0.4146987
## sample estimates:
##       cor 
## 0.3927708
